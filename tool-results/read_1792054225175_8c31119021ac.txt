     3→/**
     4→ * The Film — REEL 01, "The Twenty-Minute Tender".
     5→ *
     6→ * Six chapters, every frame model-generated (CogVideoX-3, open weights),
     7→ * sequenced by a custom cinema player: title cards, chapter ticks, a master
     8→ * clock pinned to ~02:00, 2.39:1 letterbox, grain, and a silent cut.
     9→ * No embeds, no third-party player — the reel is ours end to end.
    10→ */
    11→
    12→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
    13→import { AnimatePresence, motion, useReducedMotion } from "framer-motion";
    14→import {
    15→  Maximize,
    16→  Pause,
    17→  Play,
    18→  RotateCcw,
    19→  SkipBack,
    20→  SkipForward,
    21→  Volume,
    22→} from "lucide-react";
    23→import { cn } from "@/lib/utils";
    24→
    25→const RATE = 0.8; // deliberate, cinematic pacing
    26→const INTRO_MS = 7000;
    27→const CARD_MS = 4200;
    28→const OUTRO_MS = 10000;
    29→const NOMINAL_CLIP_MS = 12500; // 10s clip @0.8× — replaced by real metadata
    30→
    31→interface Chapter {
    32→  id: number;
    33→  title: string;
    34→  kicker: string;
    35→  line: string;
    36→  src: string;
    37→}

    84→type Phase =
    85→  | { kind: "poster" }
    86→  | { kind: "intro" }
    87→  | { kind: "card"; chapter: number }
    88→  | { kind: "clip"; chapter: number }
    89→  | { kind: "outro" };
    90→
    91→/* flat timeline segments — the master clock walks this */
    92→type Segment =
    93→  | { type: "intro"; ms: number }
    94→  | { type: "card"; chapter: number; ms: number }
    95→  | { type: "clip"; chapter: number; ms: number }
    96→  | { type: "outro"; ms: number };
    97→
    98→function fmt(ms: number) {
    99→  const s = Math.max(0, Math.round(ms / 1000));
   100→  return `${String(Math.floor(s / 60)).padStart(2, "0")}:${String(s % 60).padStart(2, "0")}`;
   101→}
   102→
   103→export function Film() {
   104→  const reduce = useReducedMotion();
   105→  const [started, setStarted] = useState(false); /* false = poster state */
   106→  const [playing, setPlaying] = useState(false);
   107→  const [elapsed, setElapsed] = useState(0);
   108→  const [durations, setDurations] = useState<Record<number, number>>({});
   109→  const [offline, setOffline] = useState<Record<number, boolean>>({});
   110→  const [posterSrc, setPosterSrc] = useState<string | null>(null);
   111→
   112→  const videoRef = useRef<HTMLVideoElement | null>(null);
   113→  const stageRef = useRef<HTMLDivElement | null>(null);
   114→  const segStart = useRef<number>(0);
   115→  const rafRef = useRef<number>(0);
   116→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   117→  const lastClip = useRef<number>(-1);
   118→
   119→  /* ── timeline ─────────────────────────────────────────────────────────── */
   120→  const segments = useMemo<Segment[]>(() => {
   121→    const segs: Segment[] = [{ type: "intro", ms: INTRO_MS }];
   122→    for (const c of CHAPTERS) {
   123→      segs.push({ type: "card", chapter: c.id, ms: CARD_MS });
   124→      segs.push({ type: "clip", chapter: c.id, ms: durations[c.id] ?? NOMINAL_CLIP_MS });
   125→    }
   126→    segs.push({ type: "outro", ms: OUTRO_MS });
   127→    return segs;
   128→  }, [durations]);
   129→
   130→  const total = useMemo(() => segments.reduce((a, s) => a + s.ms, 0), [segments]);
   131→
   132→  const segElapsedBefore = useCallback(
   133→    (idx: number) => segments.slice(0, idx).reduce((a, s) => a + s.ms, 0),
   134→    [segments]
   135→  );
   136→
   137→  const currentSegIdx = useMemo(() => {
   138→    let acc = 0;
   139→    for (let i = 0; i < segments.length; i++) {
   140→      acc += segments[i].ms;
   141→      if (elapsed < acc) return i;
   142→    }
   143→    return segments.length - 1;
   144→  }, [elapsed, segments]);
   145→
   146→  /* nominal phase — derived from the master clock, never stored */
   147→  const phase: Phase = useMemo(() => {
   148→    if (!started) return { kind: "poster" };
   149→    const seg = segments[currentSegIdx];
   150→    if (!seg) return { kind: "outro" };
   151→    if (seg.type === "intro") return { kind: "intro" };
   152→    if (seg.type === "card") return { kind: "card", chapter: seg.chapter };
   153→    if (seg.type === "clip") return { kind: "clip", chapter: seg.chapter };
   154→    return { kind: "outro" };
   155→  }, [currentSegIdx, segments, started]);
   156→
   157→  /* ── master clock ─────────────────────────────────────────────────────── */
   158→  const stopClock = useCallback(() => {
   159→    cancelAnimationFrame(rafRef.current);
   160→    if (timerRef.current) clearTimeout(timerRef.current);
   161→  }, []);
   162→
   163→  const finish = useCallback(() => {
   164→    stopClock();
   165→    setPlaying(false);
   166→    setStarted(false);
   167→    setElapsed(total);
   168→    const v = videoRef.current;
   169→    if (v) v.pause();
   170→  }, [stopClock, total]);
   171→
   172→  const tickRef = useRef<(idx: number, startedAt: number) => void>(() => undefined);
   173→  const tick = useCallback(
   174→    (idx: number, startedAt: number) => {
   175→      rafRef.current = requestAnimationFrame(() => {
   176→        const seg = segments[idx];
   177→        if (!seg) return finish();
   178→        let inSeg = performance.now() - startedAt;
   179→        if (seg.type === "clip") {
   180→          const v = videoRef.current;
   181→          const dur = durations[seg.chapter] ?? NOMINAL_CLIP_MS;
   182→          inSeg = v && !v.paused ? Math.min(dur, (v.currentTime * 1000) / RATE) : Math.min(dur, inSeg);
   183→        }
   184→        setElapsed(segElapsedBefore(idx) + Math.min(inSeg, seg.ms));
   185→        if (inSeg >= seg.ms) {
   186→          if (idx + 1 >= segments.length) return finish();
   187→          segStart.current = performance.now();
   188→          tickRef.current(idx + 1, segStart.current);
   189→          return;
   190→        }
   191→        tickRef.current(idx, startedAt);
   192→      });
   193→    },
   194→    [durations, finish, segElapsedBefore, segments]
   195→  );
   196→
   197→  useEffect(() => {
   198→    tickRef.current = tick;
   199→  }, [tick]);
   200→
   201→  const startAt = useCallback(
   202→    (idx: number) => {
   203→      stopClock();
   204→      setElapsed(segElapsedBefore(idx));
   205→      segStart.current = performance.now();
   206→      setPlaying(true);
   207→      tick(idx, segStart.current);
   208→    },
   209→    [segElapsedBefore, stopClock, tick]
   210→  );
   211→
   212→  /* ── playback control of the <video> for clip segments ────────────────── */
   213→  useEffect(() => {
   214→    const v = videoRef.current;
   215→    if (!v) return;
   216→    const seg = segments[currentSegIdx];
   217→    if (phase.kind !== "poster" && seg?.type === "clip") {
   218→      const ch = CHAPTERS.find((c) => c.id === seg.chapter);
   219→      if (ch) {
   220→        if (lastClip.current !== ch.id) {
   221→          if (!v.src.endsWith(ch.src)) v.src = ch.src;
   222→          try {
   223→            v.currentTime = 0;
   224→          } catch {
   225→            /* metadata not ready — timeupdate will settle it */
   226→          }
   227→          lastClip.current = ch.id;
   228→        }
   229→        v.playbackRate = reduce ? 1 : RATE;
   230→        if (playing) {
   231→          v.play().catch(() => {
   232→            /* autoplay guard — the clock keeps the film alive regardless */
   233→          });
   234→        } else v.pause();
   235→      }
   236→    } else {
   237→      v.pause();
   238→    }
   239→     
   240→  }, [currentSegIdx, phase.kind, playing]);
   241→
   242→  /* chapter timers for card/intro/outro are handled by the master clock */
   243→
   244→  const pause = useCallback(() => {
   245→    stopClock();
   246→    setPlaying(false);
   247→  }, [stopClock]);
   248→
   249→  const toggle = useCallback(() => {
   250→    if (!started) {
   251→      setStarted(true);
   252→      startAt(0);
   253→    } else if (playing) {
   254→      pause();
   255→    } else {
   256→      /* resume from the exact point inside the current segment */
   257→      const inSeg = elapsed - segElapsedBefore(currentSegIdx);
   258→      segStart.current = performance.now() - inSeg;
   259→      setPlaying(true);
   260→      tick(currentSegIdx, segStart.current);
   261→    }
   262→  }, [currentSegIdx, elapsed, pause, playing, segElapsedBefore, startAt, started, tick]);
   263→
   264→  const jumpChapter = useCallback(
   265→    (chapterId: number) => {
   266→      const idx = segments.findIndex((s) => "chapter" in s && s.chapter === chapterId);
   267→      if (idx >= 0) {
   268→        setStarted(true);
   269→        startAt(idx);
   270→      }
   271→    },
   272→    [segments, startAt]
   273→  );
   274→
   275→  const step = useCallback(
   276→    (dir: 1 | -1) => {
   277→      const cards = segments.map((s, i) => ({ s, i })).filter(({ s }) => s.type === "card");
   278→      if (!cards.length) return;
   279→      const firstAtOrAfter = cards.findIndex(({ i }) => i >= currentSegIdx);
   280→      const onCard = firstAtOrAfter >= 0 && cards[firstAtOrAfter].i === currentSegIdx;
   281→      let targetIdx: number;
   282→      if (dir === 1) {
   283→        if (firstAtOrAfter === -1) return; /* in the outro — nowhere forward */
   284→        targetIdx = firstAtOrAfter + (onCard ? 1 : 0);
   285→        if (targetIdx >= cards.length) return;
   286→      } else {
   287→        targetIdx =
   288→          firstAtOrAfter === -1
   289→            ? cards.length - 1
   290→            : Math.max(0, firstAtOrAfter - (onCard ? 1 : 0));
   291→      }
   292→      const seg = cards[targetIdx].s;
   293→      if (seg.type === "card") {
   294→        setStarted(true);
   295→        startAt(cards[targetIdx].i);
   296→      }
   297→    },
   298→    [currentSegIdx, segments, startAt]
   299→  );
   300→
   301→  const replay = useCallback(() => {
   302→    setStarted(true);
   303→    startAt(0);
   304→  }, [startAt]);
   305→
   306→  const fullscreen = useCallback(() => {
   307→    const el = stageRef.current;
   308→    if (!el) return;
   309→    if (document.fullscreenElement) void document.exitFullscreen();
   310→    else void el.requestFullscreen?.();
   311→  }, []);
   312→
   313→  /* keyboard */
   314→  useEffect(() => {
   315→    const onKey = (e: KeyboardEvent) => {
   316→      const tag = (e.target as HTMLElement)?.tagName;
   317→      if (tag === "INPUT" || tag === "TEXTAREA") return;
   318→      const stage = stageRef.current;
   319→      if (!stage) return;
   320→      const inView = stage.getBoundingClientRect().bottom > 80;
   321→      if (!inView) return;
   322→      if (e.code === "Space") {
   323→        e.preventDefault();
   324→        toggle();
   325→      } else if (e.code === "ArrowRight") {
   326→        const next = CHAPTERS.find((c) => segElapsedBefore(currentSegIdx) < segStartOffsetFor(c.id));
   327→        if (next) jumpChapter(next.id);
   328→      } else if (e.code === "ArrowLeft") {
   329→        const prev = [...CHAPTERS].reverse().find((c) => segStartOffsetFor(c.id) < segElapsedBefore(currentSegIdx) - 500);
   330→        if (prev) jumpChapter(prev.id);
   331→      }
   332→    };
   333→    const segStartOffsetFor = (chapterId: number) => {
   334→      const idx = segments.findIndex((s) => s.type === "card" && s.chapter === chapterId);
   335→      return idx >= 0 ? segElapsedBefore(idx) : Number.MAX_SAFE_INTEGER;
   336→    };
   337→    window.addEventListener("keydown", onKey);
   338→    return () => window.removeEventListener("keydown", onKey);
   339→     
   340→  }, [currentSegIdx, jumpChapter, segElapsedBefore, segments, toggle]);
   341→
   342→  /* poster frame — grab the true first frame of chapter one */
   343→  useEffect(() => {
   344→    const v = document.createElement("video");
   345→    v.muted = true;
   346→    v.preload = "metadata";
   347→    v.src = CHAPTERS[0].src;
   348→    const capture = () => {
   349→      try {
   350→        const canvas = document.createElement("canvas");
   351→        canvas.width = 960;
   352→        canvas.height = Math.round(960 / 2.39);
   353→        canvas.getContext("2d")?.drawImage(v, 0, 0, canvas.width, canvas.height);
   354→        setPosterSrc(canvas.toDataURL("image/jpeg", 0.82));
   355→      } catch {
   356→        /* designed poster remains */
   357→      }
   358→    };
   359→    v.addEventListener("loadeddata", capture, { once: true });
   360→    v.addEventListener("error", () => setOffline((p) => ({ ...p, 1: true })), { once: true });
   361→    return () => {
   362→      v.removeEventListener("loadeddata", capture);
   363→      v.src = "";
   364→    };
   365→  }, []);
   366→
   367→  /* probe real durations for the master clock */
   368→  useEffect(() => {
   369→    CHAPTERS.forEach((c) => {
   370→      const v = document.createElement("video");
   371→      v.preload = "metadata";
   372→      v.muted = true;
   373→      v.src = c.src;
   374→      v.addEventListener(
   375→        "loadedmetadata",
   376→        () => {
   377→          if (Number.isFinite(v.duration) && v.duration > 0) {
   378→            setDurations((p) => ({ ...p, [c.id]: (v.duration * 1000) / RATE }));
   379→          }
   380→        },
   381→        { once: true }
   382→      );
   383→      v.addEventListener("error", () => setOffline((p) => ({ ...p, [c.id]: true })), { once: true });
   384→    });
   385→  }, []);
   386→
   387→  useEffect(() => stopClock, [stopClock]);
   388→
   389→  const activeChapter =
   390→    phase.kind === "card" || phase.kind === "clip"
   391→      ? CHAPTERS.find((c) => c.id === phase.chapter)
   392→      : undefined;
   393→  const showStage = phase.kind !== "poster";
   394→  const progressPct = total > 0 ? Math.min(100, (elapsed / total) * 100) : 0;
   395→
   396→  return (
   397→    <section id="film" aria-label="The Quorum film" className="relative overflow-hidden bg-night text-white">
   398→      <div className="bg-grid-dark pointer-events-none absolute inset-0 opacity-40" aria-hidden="true" />
   399→      <div
   400→        className="pointer-events-none absolute inset-x-0 top-0 h-40 bg-gradient-to-b from-black/60 to-transparent"
   401→        aria-hidden="true"
   402→      />
   403→
   404→      <div className="relative mx-auto w-full max-w-[1200px] px-4 py-20 sm:px-6 sm:py-24 lg:px-8">
   405→        {/* header */}
   406→        <div className="flex flex-wrap items-end justify-between gap-6">
   407→          <div className="min-w-0">
   408→            <p className="flex items-center gap-2.5 font-code text-[11px] font-semibold uppercase tracking-[0.28em] text-ink-300">
   409→              <span className="inline-block h-[2px] w-6 bg-brand-500" aria-hidden="true" />
   410→              Reel 01 — Generated cinema
   411→            </p>
   412→            <h2 className="mt-4 max-w-3xl font-display text-[36px] font-black leading-[1.02] tracking-[-0.025em] sm:text-[56px]">
   413→              THE TWENTY-MINUTE
   414→              <br />
   415→              TENDER<span className="text-brand-500">.</span>
   416→            </h2>
   417→            <p className="mt-4 max-w-xl text-[14.5px] leading-relaxed text-ink-300">
   418→              A two-minute film in six chapters — every frame generated by an open-weights model, cut to the same
   419→              visual system as the product. The story of a run: the stack, the read, the evidence, the flag, the
   420→              gate, the award.
   421→            </p>
   422→          </div>
   423→          <dl className="grid shrink-0 grid-cols-2 gap-x-8 gap-y-3 font-code text-[9.5px] tracking-[0.2em] text-ink-400 sm:grid-cols-1">
   424→            {[
   425→              ["RUNTIME", `≈ ${fmt(total)}`],
   426→              ["ASPECT", "2.39 : 1"],
   427→              ["SOUND", "SILENT CUT"],
   428→              ["MODEL", "OPEN WEIGHTS"],
   429→            ].map(([k, v]) => (
   430→              <div key={k} className="flex items-baseline gap-3 sm:justify-end">
   431→                <dt className="text-ink-500">{k}</dt>
   432→                <dd className="text-[11px] font-bold text-white">{v}</dd>
   433→              </div>
   434→            ))}
   435→          </dl>
   436→        </div>
   437→
   438→        {/* stage */}
   439→        <div
   440→          ref={stageRef}
   441→          className="relative mt-10 overflow-hidden rounded-2xl border border-white/10 bg-black shadow-[0_40px_120px_rgba(0,0,0,0.6)]"
   442→        >
   443→          <div className="relative aspect-[2.39/1] w-full">
   444→            {/* designed poster — always the bedrock layer */}
   445→            <div className="absolute inset-0" aria-hidden={showStage}>
   446→              <div className="absolute inset-0 bg-[radial-gradient(120%_120%_at_50%_10%,#1a1c22_0%,#0c0d10_55%,#050506_100%)]" />
   447→              <div className="absolute inset-0 opacity-[0.16] [background-image:repeating-linear-gradient(0deg,transparent_0_2px,rgba(255,255,255,0.35)_2px_3px)]" />
   448→              <div className="absolute inset-0 grid place-items-center">
   449→                <div className="text-center">
   450→                  <p className="font-code text-[10px] tracking-[0.4em] text-ink-400">QUORUM ORIGINALS</p>
   451→                  <p className="mt-3 font-display text-[clamp(28px,6vw,72px)] font-black leading-none tracking-tight text-white">
   452→                    THE TWENTY-MINUTE
   453→                    <br />
   454→                    TENDER<span className="text-brand-500">.</span>
   455→                  </p>
   456→                  <p className="mt-4 font-code text-[10px] tracking-[0.3em] text-ink-400">
   457→                    A FILM IN SIX CHAPTERS · SHOT ENTIRELY BY MODELS
   458→                  </p>
   459→                </div>
   460→              </div>
   461→            </div>
   462→
   463→            {/* true captured poster frame */}
   464→            {posterSrc && !showStage && (
   465→              <img
   466→                src={posterSrc}
   467→                alt="First frame of the film — a records archive lit by a single shaft of light"
   468→                className="absolute inset-0 h-full w-full object-cover opacity-90"
   469→              />
   470→            )}
   471→
   472→            {/* the reel */}
   473→            {showStage && (
   474→              <>
   475→                <video
   476→                  ref={videoRef}
   477→                  className={cn(
   478→                    "absolute inset-0 h-full w-full object-cover transition-opacity duration-500",
   479→                    phase.kind === "clip" && !offline[phase.chapter] ? "opacity-100" : "opacity-0"
   480→                  )}
   481→                  muted
   482→                  playsInline
   483→                  preload="auto"
   484→                  aria-label="Film footage"
   485→                  onError={() => {
   486→                    const ch = activeChapter?.id;
   487→                    if (ch) setOffline((p) => ({ ...p, [ch]: true }));
   488→                  }}
   489→                />
   490→
   491→                {/* offline chapter interlude — designed, still on the clock */}
   492→                {phase.kind === "clip" && offline[phase.chapter] && (
   493→                  <div className="absolute inset-0 grid place-items-center bg-[radial-gradient(120%_120%_at_50%_20%,#191b21_0%,#0a0b0e_70%)]">
   494→                    <div className="text-center">
   495→                      <p className="font-code text-[10px] tracking-[0.4em] text-brand-500">REEL CHANGE — CHAPTER OFFLINE</p>
   496→                      <p className="mt-3 font-display text-[clamp(24px,4.5vw,54px)] font-black tracking-tight">
   497→                        {activeChapter?.title.toUpperCase()}
   498→                      </p>
   499→                      <p className="mt-3 max-w-md px-6 font-code text-[10.5px] leading-relaxed tracking-[0.14em] text-ink-400">
   500→                        {activeChapter?.line.toUpperCase()}
   501→                      </p>
   502→                    </div>
   503→                  </div>
   504→                )}
   505→
   506→                {/* title cards */}
   507→                <AnimatePresence mode="wait">
   508→                  {phase.kind === "intro" && (
   509→                    <motion.div
   510→                      key="intro"
   511→                      initial={{ opacity: 0 }}
   512→                      animate={{ opacity: 1 }}
   513→                      exit={{ opacity: 0, transition: { duration: 0.5 } }}
   514→                      className="absolute inset-0 grid place-items-center bg-black"
   515→                    >
   516→                      <div className="px-8 text-center">
   517→                        <motion.p
   518→                          initial={{ opacity: 0, y: 12 }}
   519→                          animate={{ opacity: 1, y: 0 }}
   520→                          transition={{ delay: 0.4, duration: 0.8 }}
   521→                          className="font-code text-[10px] tracking-[0.5em] text-brand-500"
   522→                        >
   523→                          QUORUM ORIGINALS PRESENTS
   524→                        </motion.p>
   525→                        <motion.p
   526→                          initial={{ opacity: 0, y: 16 }}
