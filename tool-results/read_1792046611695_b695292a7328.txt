   138→function useCountUp(target: number, duration = 1100): number {
   139→  const reduce = useReducedMotion();
   140→  const [val, setVal] = useState(0);
   141→  useEffect(() => {
   142→    if (reduce) {
   143→      const id = requestAnimationFrame(() => setVal(target));
   144→      return () => cancelAnimationFrame(id);
   145→    }
   146→    let raf = 0;
   147→    const t0 = performance.now();
   148→    const tick = (t: number) => {
   149→      const p = Math.min(1, (t - t0) / duration);
   150→      setVal(Math.round(target * (1 - Math.pow(1 - p, 3))));
   151→      if (p < 1) raf = requestAnimationFrame(tick);
   152→    };
   153→    raf = requestAnimationFrame(tick);
   154→    return () => cancelAnimationFrame(raf);
   155→  }, [target, duration, reduce]);
   156→  return val;
   157→}

   159→function LiveClock() {
   160→  const [now, setNow] = useState<Date | null>(null);
   161→  useEffect(() => {
   162→    const first = requestAnimationFrame(() => setNow(new Date()));
   163→    const id = setInterval(() => setNow(new Date()), 1000);
   164→    return () => {
   165→      cancelAnimationFrame(first);
   166→      clearInterval(id);
   167→    };
   168→  }, []);
   169→  return (
   170→    <span className="tabular" aria-label={now ? `Local time ${clockNow(now)}` : "Local time"}>
   171→      {now ? clockNow(now) : "--:--:--"}
   172→    </span>
   173→  );
   174→}
