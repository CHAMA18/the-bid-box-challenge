   196→function useCountUp(target: number, duration = 1100): number {
   197→  const reduce = useReducedMotion();
   198→  const [val, setVal] = useState(0);
   199→  useEffect(() => {
   200→    if (reduce) {
   201→      const id = requestAnimationFrame(() => setVal(target));
   202→      return () => cancelAnimationFrame(id);
   203→    }
   204→    let raf = 0;
   205→    const t0 = performance.now();
   206→    const tick = (t: number) => {
   207→      const p = Math.min(1, (t - t0) / duration);
   208→      setVal(Math.round(target * (1 - Math.pow(1 - p, 3))));
   209→      if (p < 1) raf = requestAnimationFrame(tick);
   210→    };
   211→    raf = requestAnimationFrame(tick);
   212→    return () => cancelAnimationFrame(raf);
   213→  }, [target, duration, reduce]);
   214→  return val;
   215→}

   217→function LiveClock() {
   218→  const [now, setNow] = useState<Date | null>(null);
   219→  useEffect(() => {
   220→    const first = requestAnimationFrame(() => setNow(new Date()));
   221→    const id = setInterval(() => setNow(new Date()), 1000);
   222→    return () => {
   223→      cancelAnimationFrame(first);
   224→      clearInterval(id);
   225→    };
   226→  }, []);
   227→  return (
   228→    <span className="tabular" aria-label={now ? `Local time ${clockNow(now)}` : "Local time"}>
   229→      {now ? clockNow(now) : "--:--:--"}
   230→    </span>
   231→  );
   232→}
