   196→   196→function useCountUp(target: number, duration = 1100): number {
   197→   197→  const reduce = useReducedMotion();
   198→   198→  const [val, setVal] = useState(0);
   199→   199→  useEffect(() => {
   200→   200→    if (reduce) {
   201→   201→      const id = requestAnimationFrame(() => setVal(target));
   202→   202→      return () => cancelAnimationFrame(id);
   203→   203→    }
   204→   204→    let raf = 0;
   205→   205→    const t0 = performance.now();
   206→   206→    const tick = (t: number) => {
   207→   207→      const p = Math.min(1, (t - t0) / duration);
   208→   208→      setVal(Math.round(target * (1 - Math.pow(1 - p, 3))));
   209→   209→      if (p < 1) raf = requestAnimationFrame(tick);
   210→   210→    };
   211→   211→    raf = requestAnimationFrame(tick);
   212→   212→    return () => cancelAnimationFrame(raf);
   213→   213→  }, [target, duration, reduce]);
   214→   214→  return val;
   215→   215→}
