   379→
   380→export function AuthScreen({ mode, onModeChange, onSuccess, onEnterWorkspace, onClose }: AuthScreenProps) {
   381→  const [values, setValues] = useState({ name: "", email: "", organization: "", password: "" });
   382→  const [remember, setRemember] = useState(true);
   383→  const [terms, setTerms] = useState(false);
   384→  const [showPw, setShowPw] = useState(false);
   385→  const [errors, setErrors] = useState<Record<string, string>>({});
   386→  const [formError, setFormError] = useState<string | null>(null);
   387→  const [status, setStatus] = useState<"form" | "sending" | "success">("form");
   388→  const [granted, setGranted] = useState<AuthUser | null>(null);
   389→  const [memoOpen, setMemoOpen] = useState(false);
   390→  const shake = useAnimationControls();
   391→
   392→  const pwScore = PW_CHECKS.filter((c) => c.test(values.password)).length;
   393→  const pwMeta = PW_META[pwScore];
   394→  const firstName = granted?.name.trim().split(/\s+/)[0] ?? null;
   395→
   396→  function set(field: keyof typeof values, v: string) {
   397→    setValues((prev) => ({ ...prev, [field]: v }));
   398→    setErrors((prev) => {
   399→      if (!prev[field]) return prev;
   400→      const next = { ...prev };
   401→      delete next[field];
   402→      return next;
   403→    });
   404→  }
   405→
   406→  /* Pre-fill the demo credentials the first time the login form shows. */
   407→  useEffect(() => {
   408→    if (mode !== "login") return;
   409→    const id = requestAnimationFrame(() => {
   410→      setValues((prev) =>
   411→        prev.email || prev.password ? prev : { ...prev, email: DEMO.email, password: DEMO.password }
   412→      );
   413→    });
   414→    return () => cancelAnimationFrame(id);
   415→  }, [mode]);
   416→
   417→  function validate(): Record<string, string> {
   418→    /* Registration validates; sign-in never blocks — see submit(). */
   419→    const e: Record<string, string> = {};
   420→    if (mode === "register") {
   421→      if (values.name.trim().length < 2) e.name = "Enter your full name.";
   422→      if (!terms) e.terms = "Please accept the terms to continue.";
   423→      if (!EMAIL_RE.test(values.email.trim())) e.email = "Enter a valid work email.";
   424→      if (values.password.length < 8) e.password = "Use at least 8 characters.";
   425→    }
   426→    return e;
   427→  }
   428→
   429→  function grant(u: AuthUser) {
   430→    setGranted(u);
   431→    setStatus("success");
   432→    onSuccess(u);
   433→  }
   434→
   435→  async function submit(ev: React.FormEvent) {
   436→    ev.preventDefault();
   437→    setFormError(null);
   438→    const errs = validate();
   439→    setErrors(errs);
   440→    if (Object.keys(errs).length > 0) {
   441→      await shake.start({ x: [0, -10, 10, -6, 6, 0], transition: { duration: 0.38 } });
   442→      return;
   443→    }
   444→
   445→    setStatus("sending");
   446→
   447→    /*
   448→     * Sign-in resolves silently: empty or malformed input falls back to the
   449→     * demo identity, and an unknown email is provisioned on the fly. The one
   450→     * honest failure is a real account with the wrong passphrase.
   451→     */
   452→    if (mode === "login") {
   453→      let email = values.email.trim();
   454→      let password = values.password;
   455→      if (!EMAIL_RE.test(email)) email = DEMO.email;
   456→      if (password.length === 0) password = DEMO.password;
   457→      setValues((prev) => ({ ...prev, email, password }));
   458→
   459→      const openLocal = () =>
   460→        grant({
   461→          id: `local-${Date.now()}`,
   462→          name: deriveName(email),
   463→          email,
   464→          organization: null,
   465→        });
   466→
   467→      try {
   468→        const res = await fetch("/api/auth/login", {
   469→          method: "POST",
   470→          headers: { "Content-Type": "application/json" },
   471→          body: JSON.stringify({ email, password, remember }),
   472→        });
   473→        const data = (await res.json().catch(() => null)) as
   474→          | { user?: AuthUser; error?: string }
   475→          | null;
   476→        if (res.ok && data?.user) {
   477→          grant(data.user);
   478→          return;
   479→        }
   480→
   481→        if (res.status === 401 && password.length >= 8) {
   482→          /* No account (yet)? Provision one silently and continue. */
   483→          const reg = await fetch("/api/auth/register", {
   484→            method: "POST",
   485→            headers: { "Content-Type": "application/json" },
   486→            body: JSON.stringify({
   487→              name: deriveName(email),
   488→              email,
   489→              password,
   490→              organization: email === DEMO.email ? DEMO.organization : undefined,
   491→            }),
   492→          });
   493→          if (reg.ok) {
   494→            const rd = (await reg.json().catch(() => null)) as { user?: AuthUser } | null;
   495→            if (rd?.user) {
   496→              grant(rd.user);
   497→              return;
   498→            }
   499→          }
   500→          if (reg.status === 409) {
   501→            /* A real account exists and the passphrase didn't match. */
   502→            setFormError(
