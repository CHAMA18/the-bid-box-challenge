   381→export function AuthScreen({ mode, onModeChange, onSuccess, onEnterWorkspace, onClose }: AuthScreenProps) {
   382→  const [values, setValues] = useState({ name: "", email: "", organization: "", password: "" });
   383→  const [remember, setRemember] = useState(true);
   384→  const [terms, setTerms] = useState(false);
   385→  const [showPw, setShowPw] = useState(false);
   386→  const [errors, setErrors] = useState<Record<string, string>>({});
   387→  const [formError, setFormError] = useState<string | null>(null);
   388→  const [status, setStatus] = useState<"form" | "sending" | "success">("form");
   389→  const [granted, setGranted] = useState<AuthUser | null>(null);
   390→  const [memoOpen, setMemoOpen] = useState(false);
   391→  const shake = useAnimationControls();
   392→
   393→  const pwScore = PW_CHECKS.filter((c) => c.test(values.password)).length;
   394→  const pwMeta = PW_META[pwScore];
   395→  const firstName = granted?.name.trim().split(/\s+/)[0] ?? null;
   396→
   397→  function set(field: keyof typeof values, v: string) {
   398→    setValues((prev) => ({ ...prev, [field]: v }));
   399→    setErrors((prev) => {
   400→      if (!prev[field]) return prev;
   401→      const next = { ...prev };
   402→      delete next[field];
   403→      return next;
   404→    });
   405→  }
   406→
   407→  /* Pre-fill the demo credentials the first time the login form shows. */
   408→  useEffect(() => {
   409→    if (mode !== "login") return;
   410→    const id = requestAnimationFrame(() => {
   411→      setValues((prev) =>
   412→        prev.email || prev.password ? prev : { ...prev, email: DEMO.email, password: DEMO.password }
   413→      );
   414→    });
   415→    return () => cancelAnimationFrame(id);
   416→  }, [mode]);
   417→
   418→  function validate(): Record<string, string> {
   419→    /* Registration validates; sign-in never blocks — see submit(). */
   420→    const e: Record<string, string> = {};
   421→    if (mode === "register") {
   422→      if (values.name.trim().length < 2) e.name = "Enter your full name.";
   423→      if (!terms) e.terms = "Please accept the terms to continue.";
   424→      if (!EMAIL_RE.test(values.email.trim())) e.email = "Enter a valid work email.";
   425→      if (values.password.length < 8) e.password = "Use at least 8 characters.";
   426→    }
   427→    return e;
   428→  }
   429→
   430→  function grant(u: AuthUser) {
   431→    setGranted(u);
   432→    setStatus("success");
   433→    onSuccess(u);
   434→  }
   435→
   436→  async function submit(ev: React.FormEvent) {
   437→    ev.preventDefault();
   438→    setFormError(null);
   439→    const errs = validate();
   440→    setErrors(errs);
   441→    if (Object.keys(errs).length > 0) {
   442→      await shake.start({ x: [0, -10, 10, -6, 6, 0], transition: { duration: 0.38 } });
   443→      return;
   444→    }
   445→
   446→    setStatus("sending");
   447→
   448→    /*
   449→     * Sign-in resolves silently: empty or malformed input falls back to the
   450→     * demo identity, and an unknown email is provisioned on the fly. The one
   451→     * honest failure is a real account with the wrong passphrase.
   452→     */
   453→    if (mode === "login") {
   454→      let email = values.email.trim();
   455→      let password = values.password;
   456→      if (!EMAIL_RE.test(email)) email = DEMO.email;
   457→      if (password.length === 0) password = DEMO.password;
   458→      setValues((prev) => ({ ...prev, email, password }));
   459→
   460→      const openLocal = () => {
   461→        /* Mirror the demo identity into Firebase Auth too (best-effort). */
   462→        void syncFirebaseIdentity({ name: deriveName(email), email, password, mode: "login" });
   463→        grant({
   464→          id: `local-${Date.now()}`,
   465→          name: deriveName(email),
   466→          email,
   467→          organization: null,
   468→        });
   469→      };
   470→
   471→      try {
   472→        const res = await fetch("/api/auth/login", {
   473→          method: "POST",
   474→          headers: { "Content-Type": "application/json" },
   475→          body: JSON.stringify({ email, password, remember }),
   476→        });
   477→        const data = (await res.json().catch(() => null)) as
   478→          | { user?: AuthUser; error?: string }
   479→          | null;
   480→        if (res.ok && data?.user) {
   481→          void syncFirebaseIdentity({ name: data.user.name, email, password, mode: "login" });
   482→          grant(data.user);
   483→          return;
   484→        }
   485→
   486→        if (res.status === 401 && password.length >= 8) {
   487→          /* No account (yet)? Provision one silently and continue. */
   488→          const reg = await fetch("/api/auth/register", {
   489→            method: "POST",
   490→            headers: { "Content-Type": "application/json" },
   491→            body: JSON.stringify({
   492→              name: deriveName(email),
   493→              email,
   494→              password,
   495→              organization: email === DEMO.email ? DEMO.organization : undefined,
   496→            }),
   497→          });
   498→          if (reg.ok) {
   499→            const rd = (await reg.json().catch(() => null)) as { user?: AuthUser } | null;
