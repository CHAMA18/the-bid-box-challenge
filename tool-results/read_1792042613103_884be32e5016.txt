   315→export function AuthScreen({ mode, onModeChange, onSuccess, onClose }: AuthScreenProps) {
   316→  const [values, setValues] = useState({ name: "", email: "", organization: "", password: "" });
   317→  const [remember, setRemember] = useState(true);
   318→  const [terms, setTerms] = useState(false);
   319→  const [showPw, setShowPw] = useState(false);
   320→  const [errors, setErrors] = useState<Record<string, string>>({});
   321→  const [formError, setFormError] = useState<string | null>(null);
   322→  const [status, setStatus] = useState<"form" | "sending" | "success">("form");
   323→  const [granted, setGranted] = useState<AuthUser | null>(null);
   324→  const [memoOpen, setMemoOpen] = useState(false);
   325→  const shake = useAnimationControls();
   326→
   327→  const pwScore = PW_CHECKS.filter((c) => c.test(values.password)).length;
   328→  const pwMeta = PW_META[pwScore];
   329→  const firstName = granted?.name.trim().split(/\s+/)[0] ?? null;
   330→
   331→  function set(field: keyof typeof values, v: string) {
   332→    setValues((prev) => ({ ...prev, [field]: v }));
   333→    setErrors((prev) => {
   334→      if (!prev[field]) return prev;
   335→      const next = { ...prev };
   336→      delete next[field];
   337→      return next;
   338→    });
   339→  }
   340→
   341→  function validate(): Record<string, string> {
   342→    const e: Record<string, string> = {};
   343→    if (mode === "register") {
   344→      if (values.name.trim().length < 2) e.name = "Enter your full name.";
   345→      if (!terms) e.terms = "Please accept the terms to continue.";
   346→    }
   347→    if (!EMAIL_RE.test(values.email.trim())) e.email = "Enter a valid work email.";
   348→    if (mode === "register" && values.password.length < 8) e.password = "Use at least 8 characters.";
   349→    if (mode === "login" && values.password.length === 0) e.password = "Enter your password.";
   350→    return e;
   351→  }
   352→
   353→  async function submit(ev: React.FormEvent) {
   354→    ev.preventDefault();
   355→    setFormError(null);
   356→    const errs = validate();
   357→    setErrors(errs);
   358→    if (Object.keys(errs).length > 0) {
   359→      await shake.start({ x: [0, -10, 10, -6, 6, 0], transition: { duration: 0.38 } });
   360→      return;
   361→    }
   362→
   363→    setStatus("sending");
   364→    try {
   365→      const res = await fetch(`/api/auth/${mode}`, {
   366→        method: "POST",
   367→        headers: { "Content-Type": "application/json" },
   368→        body: JSON.stringify(
   369→          mode === "register"
   370→            ? {
   371→                name: values.name.trim(),
   372→                email: values.email.trim(),
   373→                password: values.password,
   374→                organization: values.organization.trim() || undefined,
   375→              }
   376→            : { email: values.email.trim(), password: values.password, remember }
   377→        ),
   378→      });
   379→      const data = (await res.json().catch(() => null)) as
   380→        | { user?: AuthUser; error?: string; errors?: Record<string, string> }
   381→        | null;
   382→
   383→      if (!res.ok || !data?.user) {
   384→        if (data?.errors) setErrors(data.errors);
   385→        setFormError(data?.error ?? "Something went wrong on our side. Try again.");
   386→        setStatus("form");
   387→        return;
   388→      }
   389→
   390→      setGranted(data.user);
   391→      setStatus("success");
   392→      onSuccess(data.user);
   393→    } catch {
   394→      setFormError("Cannot reach the server. Check your connection and try again.");
   395→      setStatus("form");
   396→    }
   397→  }
   398→
   399→  return (
   400→    <div className="flex min-h-screen flex-col bg-paper font-body text-ink-900 antialiased">
   401→      {/* Top bar */}
   402→      <header className="relative z-10 flex items-center justify-between gap-4 border-b border-white/10 bg-night px-4 py-3.5 sm:px-8">
   403→        <button
   404→          type="button"
   405→          onClick={onClose}
   406→          aria-label="Back to the Bid Box site"
   407→          className="rounded-lg outline-none transition-opacity hover:opacity-85 focus-visible:ring-2 focus-visible:ring-brand-500"
   408→        >
   409→          <BidBoxLogo variant="light" />
   410→        </button>
   411→        <div className="flex items-center gap-4 sm:gap-6">
   412→          <p className="hidden items-center gap-2.5 font-code text-[9px] tracking-[0.22em] text-ink-400 md:flex">
   413→            <span className="relative flex h-1.5 w-1.5">
   414→              <span
   415→                className="absolute inline-flex h-full w-full animate-ping-slow rounded-full bg-brand-500"
   416→                aria-hidden="true"
   417→              />
   418→              <span className="relative inline-flex h-1.5 w-1.5 rounded-full bg-brand-600" aria-hidden="true" />
   419→            </span>
   420→            SECURE · SCRYPT-HASHED · HTTPONLY SESSION
   421→          </p>
   422→          <button
   423→            type="button"
   424→            onClick={onClose}
   425→            className="inline-flex h-9 shrink-0 items-center gap-2 whitespace-nowrap rounded-full border border-white/15 px-4 text-[12.5px] font-semibold text-ink-200 transition-all hover:border-white/40 hover:text-white"
   426→          >
   427→            <ArrowLeft className="h-3.5 w-3.5" aria-hidden="true" />
   428→            Back to site
   429→          </button>
   430→        </div>
   431→      </header>

   433→      <div className="grid flex-1 lg:grid-cols-2">
   434→        <BrandPanel grantedFirstName={firstName} />
   435→
   436→        {/* Form column */}
   437→        <main className="relative flex flex-col bg-paper">
   438→          <BrandStrip />
   439→
   440→          <div className="flex flex-1 items-center justify-center px-4 py-12 sm:px-10 lg:py-16">

   445→                ) : (

   453→                    <Eyebrow>
