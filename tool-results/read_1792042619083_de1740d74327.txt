   315→   315→export function AuthScreen({ mode, onModeChange, onSuccess, onClose }: AuthScreenProps) {
   316→   316→  const [values, setValues] = useState({ name: "", email: "", organization: "", password: "" });
   317→   317→  const [remember, setRemember] = useState(true);
   318→   318→  const [terms, setTerms] = useState(false);
   319→   319→  const [showPw, setShowPw] = useState(false);
   320→   320→  const [errors, setErrors] = useState<Record<string, string>>({});
   321→   321→  const [formError, setFormError] = useState<string | null>(null);
   322→   322→  const [status, setStatus] = useState<"form" | "sending" | "success">("form");
   323→   323→  const [granted, setGranted] = useState<AuthUser | null>(null);
   324→   324→  const [memoOpen, setMemoOpen] = useState(false);
   325→   325→  const shake = useAnimationControls();
   326→   326→
   327→   327→  const pwScore = PW_CHECKS.filter((c) => c.test(values.password)).length;
   328→   328→  const pwMeta = PW_META[pwScore];
   329→   329→  const firstName = granted?.name.trim().split(/\s+/)[0] ?? null;
   330→   330→
   331→   331→  function set(field: keyof typeof values, v: string) {
   332→   332→    setValues((prev) => ({ ...prev, [field]: v }));
   333→   333→    setErrors((prev) => {
   334→   334→      if (!prev[field]) return prev;
   335→   335→      const next = { ...prev };
   336→   336→      delete next[field];
   337→   337→      return next;
   338→   338→    });
   339→   339→  }
   340→   340→
   341→   341→  function validate(): Record<string, string> {
   342→   342→    const e: Record<string, string> = {};
   343→   343→    if (mode === "register") {
   344→   344→      if (values.name.trim().length < 2) e.name = "Enter your full name.";
   345→   345→      if (!terms) e.terms = "Please accept the terms to continue.";
   346→   346→    }
   347→   347→    if (!EMAIL_RE.test(values.email.trim())) e.email = "Enter a valid work email.";
   348→   348→    if (mode === "register" && values.password.length < 8) e.password = "Use at least 8 characters.";
   349→   349→    if (mode === "login" && values.password.length === 0) e.password = "Enter your password.";
   350→   350→    return e;
   351→   351→  }
   352→   352→
   353→   353→  async function submit(ev: React.FormEvent) {
   354→   354→    ev.preventDefault();
   355→   355→    setFormError(null);
   356→   356→    const errs = validate();
   357→   357→    setErrors(errs);
   358→   358→    if (Object.keys(errs).length > 0) {
   359→   359→      await shake.start({ x: [0, -10, 10, -6, 6, 0], transition: { duration: 0.38 } });
   360→   360→      return;
   361→   361→    }
   362→   362→
   363→   363→    setStatus("sending");
   364→   364→    try {
   365→   365→      const res = await fetch(`/api/auth/${mode}`, {
   366→   366→        method: "POST",
   367→   367→        headers: { "Content-Type": "application/json" },
   368→   368→        body: JSON.stringify(
   369→   369→          mode === "register"
   370→   370→            ? {
   371→   371→                name: values.name.trim(),
   372→   372→                email: values.email.trim(),
   373→   373→                password: values.password,
   374→   374→                organization: values.organization.trim() || undefined,
   375→   375→              }
   376→   376→            : { email: values.email.trim(), password: values.password, remember }
   377→   377→        ),
   378→   378→      });
   379→   379→      const data = (await res.json().catch(() => null)) as
   380→   380→        | { user?: AuthUser; error?: string; errors?: Record<string, string> }
   381→   381→        | null;
   382→   382→
   383→   383→      if (!res.ok || !data?.user) {
   384→   384→        if (data?.errors) setErrors(data.errors);
   385→   385→        setFormError(data?.error ?? "Something went wrong on our side. Try again.");
   386→   386→        setStatus("form");
   387→   387→        return;
   388→   388→      }
   389→   389→
   390→   390→      setGranted(data.user);
   391→   391→      setStatus("success");
   392→   392→      onSuccess(data.user);
   393→   393→    } catch {
   394→   394→      setFormError("Cannot reach the server. Check your connection and try again.");
   395→   395→      setStatus("form");
   396→   396→    }
   397→   397→  }
   398→   398→
   399→   399→  return (
   400→   400→    <div className="flex min-h-screen flex-col bg-paper font-body text-ink-900 antialiased">
   401→   401→      {/* Top bar */}
   402→   402→      <header className="relative z-10 flex items-center justify-between gap-4 border-b border-white/10 bg-night px-4 py-3.5 sm:px-8">
   403→   403→        <button
   404→   404→          type="button"
   405→   405→          onClick={onClose}
   406→   406→          aria-label="Back to the Bid Box site"
   407→   407→          className="rounded-lg outline-none transition-opacity hover:opacity-85 focus-visible:ring-2 focus-visible:ring-brand-500"
   408→   408→        >
   409→   409→          <BidBoxLogo variant="light" />
